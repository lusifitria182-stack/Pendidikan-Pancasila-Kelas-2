⚡ Integrasi Google Apps Script & Google Docs

Guru dapat menghubungkan media pembelajaran ini dengan Google Docs milik sekolah menggunakan script Google Apps Script gratis.

Klik di sini untuk melihat Kode Google Apps Script (Code.gs) 
// Paste kode ini di script.google.com dan deploy sebagai Web App (Anyone)
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  
  // Buat / Buka Google Doc Laporan
  var doc = DocumentApp.create("Laporan - " + data.nama + " (" + data.tanggal + ")");
  var body = doc.getBody();
  
  body.appendParagraph("LAPORAN HASIL BELAJAR PANCASILA").setHeading(DocumentApp.ParagraphHeading.HEADING1);
  body.appendParagraph("SD Negeri 028 Senayang");
  body.appendParagraph("------------------------------------------------");
  body.appendParagraph("Nama Siswa: " + data.nama);
  body.appendParagraph("Nilai Kuis: " + data.nilaiKuis);
  body.appendParagraph("Status Game: " + data.statusGame);
  body.appendParagraph("Total Bintang: " + data.bintang);
  body.appendParagraph("Tanggal: " + data.tanggal);
  
  return ContentService.createTextOutput(JSON.stringify({result: "success", url: doc.getUrl()}))
    .setMimeType(ContentService.MimeType.JSON);
}
